    Nginx 보안 설정¶

    서버 정보 숨기기¶

    server_tokens off¶

    Nginx 기본 설정에서는 응답 헤더에 Server: nginx/1.24.0처럼 버전이 찍힌다. 에러 페이지(404, 502 등)에도 하단에 버전이 노출된다. 공격자가 버전별 알려진 CVE를 바로 확인할 수 있으므로 반드시 꺼야 한다.

    # http 블록에 설정 (전체 적용)
http {
    server_tokens off;
    # Server: nginx/1.24.0 → Server: nginx
}

    server_tokens off를 해도 Server: nginx는 남는다. 이것까지 지우려면 headers-more-nginx-module을 써야 한다.

    # headers-more 모듈 사용 시
more_clear_headers Server;
# 또는
more_set_headers "Server: custom";

    Ubuntu에서 모듈 설치:

    sudo apt install nginx-extras
# nginx-extras에 headers-more 모듈이 포함되어 있다

    불필요한 HTTP 메서드 차단¶

    GET, POST, PUT, DELETE 외에 TRACE, OPTIONS 같은 메서드가 열려 있으면 정보 유출 경로가 된다. TRACE는 XST(Cross-Site Tracing) 공격에 쓸 수 있다.

    server {
    # 허용할 메서드만 명시
    if ($request_method !~ ^(GET|POST|PUT|DELETE|PATCH)$) {
        return 405;
    }
}

    민감 경로 차단¶

    .git, .env, .htpasswd 등 숨김 파일 차단¶

    배포할 때 .git 디렉토리가 웹 루트에 같이 올라가는 경우가 있다. .git/config에 리모트 URL이, .git/HEAD에 브랜치 정보가 들어있다. .env에는 DB 비밀번호, API 키가 들어있는 경우가 많다.

    server {
    # 점(.)으로 시작하는 숨김 파일/디렉토리 전부 차단
    location ~ /\. {
        deny all;
        return 404;
    }
}

    이 한 줄로 .git, .env, .htpasswd, .svn, .DS_Store 등을 전부 막는다. 다만 Let's Encrypt 인증 경로 .well-known도 막히므로 예외 처리가 필요하다.

    server {
    # Let's Encrypt 인증 경로는 허용
    location ~ /\.well-known {
        allow all;
    }

    # 나머지 숨김 파일은 차단
    location ~ /\. {
        deny all;
        return 404;
    }
}

    location 블록은 위에서 아래로 매칭되는 게 아니라 정규식 매칭 우선순위를 따른다. 같은 ~ 타입이면 설정 파일에서 먼저 나온 쪽이 우선이므로 .well-known을 . 차단보다 위에 둬야 한다.

    백업 파일, 설정 파일 차단¶

    에디터가 만드는 백업 파일이나 설정 파일이 웹으로 노출되는 경우도 있다.

    server {
    # 백업 파일
    location ~* \.(bak|old|orig|save|swp|swo|tmp)$ {
        deny all;
        return 404;
    }

    # 소스 코드, 설정 파일
    location ~* \.(sql|log|ini|conf|yml|yaml|toml|xml|properties)$ {
        deny all;
        return 404;
    }
}

    특정 경로 명시적 차단¶

    server {
    # WordPress 공격 시도 차단 (WordPress를 쓰지 않는 경우)
    location ~ ^/(wp-admin|wp-login|xmlrpc\.php) {
        deny all;
        return 404;
    }

    # phpMyAdmin 스캔 차단
    location ~ ^/(phpmyadmin|pma|myadmin) {
        deny all;
        return 404;
    }
}

    자동화 스캐너가 /wp-login.php, /phpmyadmin 같은 경로를 무차별로 때리는 건 로그를 보면 매일 확인할 수 있다. 해당 서비스를 쓰지 않으면 아예 404로 떨궈서 로그도 줄이고 불필요한 처리도 막는다.

    IP 기반 접근 제어¶

    allow / deny¶

    server {
    # 관리자 페이지 — 사내 IP만 허용
    location /admin/ {
        allow 10.0.0.0/8;         # 사내망
        allow 172.16.0.0/12;      # VPN
        allow 192.168.1.100;      # 특정 IP
        deny all;                 # 나머지 전부 차단

        proxy_pass http://backend;
    }
}

    allow/deny는 위에서부터 순서대로 평가한다. 첫 번째로 매칭되는 규칙이 적용되므로 allow를 먼저 쓰고 마지막에 deny all을 넣는다.

    전체 서버에 IP 제한 걸기¶

    # 내부 관리용 서버 — 사내에서만 접근
server {
    listen 443 ssl;
    server_name admin.example.com;

    allow 10.0.0.0/8;
    deny all;

    location / {
        proxy_pass http://admin-backend:9000;
    }
}

    Spring Boot Actuator 보호¶

    Actuator 엔드포인트(/actuator/env, /actuator/configprops 등)는 환경 변수, DB 접속 정보가 그대로 노출된다. 외부 접근을 반드시 막아야 한다.

    server {
    # Actuator 전체 차단 (외부)
    location /actuator/ {
        allow 10.0.0.0/8;     # 모니터링 서버만 허용
        deny all;
    }

    # 헬스 체크만 열어두는 경우
    location = /actuator/health {
        proxy_pass http://backend;
    }

    location /actuator/ {
        deny all;
    }
}

    geo 모듈로 국가/대역 차단¶

    http {
    geo $blocked_ip {
        default 0;
        # 특정 대역 차단
        185.220.0.0/16 1;    # 알려진 공격 대역
        45.155.0.0/16 1;
    }

    server {
        if ($blocked_ip) {
            return 403;
        }
    }
}

    Basic Auth¶

    관리 페이지나 스테이징 환경에 간단한 인증을 걸 때 쓴다. 프로덕션 인증 체계를 대체하는 용도가 아니라, 추가 방어막으로 쓴다.

    비밀번호 파일 생성¶

    # htpasswd 설치
sudo apt install apache2-utils

# 사용자 추가 (-c: 파일 생성, 두 번째부터는 -c 빼야 한다)
sudo htpasswd -c /etc/nginx/.htpasswd admin
# 비밀번호 입력 프롬프트

# 사용자 추가
sudo htpasswd /etc/nginx/.htpasswd devuser

# 파일 내용 확인 (bcrypt 해시)
cat /etc/nginx/.htpasswd
# admin:$apr1$xyz...
# devuser:$apr1$abc...

# 파일 권한 설정
sudo chmod 640 /etc/nginx/.htpasswd
sudo chown root:www-data /etc/nginx/.htpasswd

    -c 옵션을 두 번째 사용자 추가할 때도 쓰면 파일이 덮어써져서 기존 사용자가 날아간다. 실수하기 쉬운 부분이다.

    Nginx 설정¶

    server {
    # 스테이징 환경 전체에 Basic Auth
    auth_basic "Restricted";
    auth_basic_user_file /etc/nginx/.htpasswd;

    location / {
        proxy_pass http://backend;
    }

    # 헬스 체크는 인증 제외
    location = /health {
        auth_basic off;
        proxy_pass http://backend;
    }

    # API는 자체 인증이 있으므로 Basic Auth 제외
    location /api/ {
        auth_basic off;
        proxy_pass http://backend;
    }
}

    IP 제한 + Basic Auth 조합¶

    server {
    location /admin/ {
        # 사내 IP면 인증 없이 접근
        satisfy any;

        allow 10.0.0.0/8;
        deny all;

        auth_basic "Admin Area";
        auth_basic_user_file /etc/nginx/.htpasswd;

        proxy_pass http://backend;
    }
}

    satisfy any는 IP 허용 또는 인증 성공 중 하나만 만족하면 통과한다. satisfy all(기본값)로 바꾸면 IP 허용 그리고 인증 성공 둘 다 필요하다.

    요청 크기와 속도 제한 조합¶

    Rate Limiting 단독으로는 대용량 파일을 느리게 올리는 공격이나 연결만 잡고 놓지 않는 slowloris 공격을 막을 수 없다. 크기 제한, 속도 제한, 연결 수 제한을 함께 걸어야 한다.

    요청 처리 흐름¶

    세 가지 방어 계층이 요청을 어떤 순서로 검사하는지 파악해야 설정 간 충돌을 피할 수 있다.

    클라이언트 요청 도착
        │
        ▼
┌───────────────────────────┐
│ 1) 연결 수 검사           │
│    limit_conn             │
│    (IP당 동시 연결 제한)  │
└─────────┬─────────────────┘
          │ 연결 허용
          ▼
┌───────────────────────────┐
│ 2) 요청 속도 검사         │
│    limit_req              │
│    (초당/분당 요청 수)    │
└─────────┬─────────────────┘
          │ rate 범위 내
          ▼
┌───────────────────────────┐
│ 3) 헤더 수신              │
│    client_header_timeout  │──── 타임아웃 → 408 반환
│    (Slowloris 방어)       │     (느린 헤더 전송 차단)
└─────────┬─────────────────┘
          │ 헤더 수신 완료
          ▼
┌───────────────────────────┐
│ 4) 요청 크기 검사         │
│    client_max_body_size   │──── 초과 → 413 반환
│    (바디 크기 제한)       │
└─────────┬─────────────────┘
          │ 크기 허용
          ▼
┌───────────────────────────┐
│ 5) 바디 수신              │
│    client_body_timeout    │──── 타임아웃 → 408 반환
│    (느린 바디 전송 차단)  │
└─────────┬─────────────────┘
          │ 수신 완료
          ▼
     백엔드 전달 (proxy_pass)

    각 단계에서 거부되면 이후 단계는 실행되지 않는다. limit_conn에서 걸리면 limit_req 검사 자체가 발생하지 않으므로, 로그 분석할 때 어떤 단계에서 거부됐는지 확인해야 한다.

    설정 예시¶

    http {
    # 속도 제한
    limit_req_zone $binary_remote_addr zone=upload:10m rate=5r/m;
    limit_conn_zone $binary_remote_addr zone=conn:10m;
}

server {
    # 기본 요청 크기 제한
    client_max_body_size 10m;

    # 파일 업로드 엔드포인트
    location /api/upload {
        client_max_body_size 50m;    # 업로드만 50MB 허용

        # 분당 5회 제한
        limit_req zone=upload burst=2;

        # IP당 동시 연결 2개 제한
        limit_conn conn 2;

        proxy_pass http://backend;
    }

    # 일반 API
    location /api/ {
        client_max_body_size 1m;     # JSON 요청은 1MB면 충분

        proxy_pass http://backend;
    }
}

    Slowloris 방어¶

    Slowloris는 연결을 맺은 뒤 헤더를 아주 천천히 보내서 worker 연결을 점유하는 공격이다.

    http {
    # 클라이언트 헤더 읽기 타임아웃 (기본 60s → 줄인다)
    client_header_timeout 10s;

    # 클라이언트 바디 읽기 타임아웃
    client_body_timeout 10s;

    # 클라이언트에게 응답 보내기 타임아웃
    send_timeout 10s;

    # 요청 헤더 크기 제한
    large_client_header_buffers 4 8k;
}

    기본 60초는 너무 넉넉하다. 정상 클라이언트는 헤더 전송에 10초도 안 걸린다. 다만 대용량 파일 업로드가 있는 엔드포인트에서는 client_body_timeout을 별도로 늘려야 한다.

    DDoS 공격 유형별 Nginx 방어 설정¶

    공격 유형마다 Nginx에서 대응하는 설정이 다르다. 한 가지 설정으로 모든 공격을 막을 수 없으므로, 공격 특성에 맞는 설정을 조합해야 한다.

    공격 유형 공격 방식 Nginx 방어 설정 비고
    HTTP Flood 대량의 정상 HTTP 요청 limit_req (요청 속도 제한) burst와 nodelay 조합으로 순간 트래픽 허용 범위 조정
    Slowloris 헤더를 천천히 보내 연결 점유 client_header_timeout 10s worker_connections 소진이 목표
    Slow POST 바디를 천천히 보내 연결 점유 client_body_timeout 10s Slowloris와 원리 동일, 바디 타겟
    Slow Read 응답을 천천히 읽어 연결 점유 send_timeout 10s 클라이언트가 수신 윈도우를 줄임
    Connection Flood 대량 TCP 연결 생성 limit_conn (동시 연결 수 제한) OS 레벨 connlimit와 병행
    Large Payload 큰 요청 바디로 리소스 소진 client_max_body_size 엔드포인트별 차등 설정 필요
    헤더 폭탄 비정상적으로 큰 헤더 전송 large_client_header_buffers 4 8k 기본 4개 × 8KB, 초과 시 400 반환
    Brute Force 로그인 등 특정 엔드포인트 반복 limit_req (엄격한 rate) + fail2ban rate=5r/m 수준으로 강하게 제한
    봇 트래픽 자동화 도구로 대량 요청 map $http_user_agent 차단 User-Agent 기반, 우회 가능성 있음
    IP 대역 공격 특정 대역에서 집중 공격 geo + deny 알려진 공격 대역 사전 차단

    Nginx 단독으로 방어가 어려운 공격:

    • SYN Flood: TCP 레벨 공격이므로 Nginx까지 도달하지 않는다. OS 커널(net.ipv4.tcp_syncookies)이나 방화벽에서 처리해야 한다.
    • UDP Flood / DNS Amplification: Nginx는 HTTP 서버이므로 UDP 기반 공격과 무관하다. 네트워크 장비나 클라우드 DDoS 방어 서비스가 필요하다.
    • 분산 저속 공격: 수천 개 IP에서 각각 rate 미만으로 요청하면 limit_req로 잡히지 않는다. WAF나 행동 분석 기반 솔루션이 필요하다.

    대규모 DDoS에 대비하려면 Nginx 설정만으로는 한계가 있다. 앞단에 CDN(Cloudflare, AWS CloudFront)이나 클라우드 DDoS 방어 서비스를 두고, Nginx는 CDN을 통과한 트래픽에 대한 2차 방어선으로 운영하는 구조가 현실적이다.

    악성 봇 / User-Agent 차단¶

    알려진 악성 봇 차단¶

    http {
    map $http_user_agent $bad_bot {
        default 0;
        ~*(?:sqlmap|nikto|nmap|masscan|dirbuster|gobuster) 1;  # 스캐너
        ~*(?:scrapy|python-requests|Go-http-client) 1;         # 크롤러
        ~*(?:curl|wget) 1;                                      # CLI 도구
        "" 1;                                                    # User-Agent 없음
    }
}

server {
    if ($bad_bot) {
        return 403;
    }
}

    python-requests나 curl을 막을지는 서비스 성격에 따라 다르다. API 서버에서 정상적인 클라이언트가 이 User-Agent를 쓰는 경우가 있으므로 무조건 막으면 안 된다. 웹사이트 서빙 서버에서만 쓰는 것을 권장한다.

    빈 User-Agent 차단¶

    User-Agent가 비어있는 요청은 대부분 봇이다. 정상 브라우저는 항상 User-Agent를 보낸다.

    server {
    if ($http_user_agent = "") {
        return 403;
    }
}

    Referer 기반 핫링크 방지¶

    외부에서 이미지나 파일을 직접 링크하는 것을 막는다.

    server {
    location ~* \.(jpg|jpeg|png|gif|webp|mp4)$ {
        valid_referers none blocked server_names *.example.com;
        if ($invalid_referer) {
            return 403;
        }
    }
}

    디렉토리 트래버설 방어¶

    기본 방어¶

    공격자가 /../../../etc/passwd 같은 경로로 상위 디렉토리의 파일을 읽으려는 시도다. Nginx는 기본적으로 URI를 정규화해서 ..을 제거하지만, 인코딩을 이용한 우회가 있을 수 있다.

    server {
    # URI에 ..이 포함된 요청 차단
    if ($request_uri ~* "\.\.") {
        return 403;
    }

    # null byte injection 차단
    if ($request_uri ~* "\x00") {
        return 403;
    }
}

    alias와 root의 차이 — 트래버설 원인이 되는 설정 실수¶

    # 잘못된 설정 — 트래버설 가능
location /files {
    alias /var/www/uploads/;
}
# /files../etc/passwd → /var/www/uploads/../etc/passwd → /var/www/etc/passwd

# 올바른 설정 — location 끝에 / 붙이기
location /files/ {
    alias /var/www/uploads/;
}

    alias를 쓸 때 location에 trailing slash가 없으면 ..을 이용한 경로 탈출이 가능하다. root를 쓰면 이 문제가 없으므로 가능하면 root를 쓰고, alias가 필요하면 반드시 location 끝에 /를 붙인다.

    autoindex 비활성화¶

    server {
    autoindex off;  # 기본값이 off이지만 명시적으로 확인
}

    autoindex on으로 설정하면 index 파일이 없는 디렉토리에 접근할 때 파일 목록이 보인다. 개발 환경에서 켜놓고 프로덕션에 그대로 올리는 실수가 있다.

    종합 보안 설정 예시¶

    위 내용을 스니펫으로 정리해서 include로 재사용한다.

    # /etc/nginx/snippets/security.conf

# 서버 정보 숨기기
server_tokens off;

# 타임아웃
client_header_timeout 10s;
client_body_timeout 10s;
send_timeout 10s;

# 요청 크기 기본값
client_max_body_size 10m;
large_client_header_buffers 4 8k;

    # /etc/nginx/sites-available/example.com

server {
    listen 443 ssl http2;
    server_name example.com;

    include snippets/ssl-params.conf;
    include snippets/security.conf;

    # 숨김 파일 차단
    location ~ /\.well-known { allow all; }
    location ~ /\. { deny all; return 404; }

    # 백업/설정 파일 차단
    location ~* \.(bak|old|orig|save|swp|sql|log)$ {
        deny all;
        return 404;
    }

    # HTTP 메서드 제한
    if ($request_method !~ ^(GET|POST|PUT|DELETE|PATCH)$) {
        return 405;
    }

    # 디렉토리 트래버설 방어
    if ($request_uri ~* "\.\.") {
        return 403;
    }

    # 프론트엔드
    location / {
        root /var/www/frontend/dist;
        try_files $uri $uri/ /index.html;
    }

    # API
    location /api/ {
        proxy_pass http://backend;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
    }

    # 관리자 — IP + Basic Auth
    location /admin/ {
        satisfy any;
        allow 10.0.0.0/8;
        deny all;
        auth_basic "Admin";
        auth_basic_user_file /etc/nginx/.htpasswd;

        proxy_pass http://backend;
    }

    # Actuator 차단
    location /actuator/ {
        allow 10.0.0.0/8;
        deny all;
    }
}
