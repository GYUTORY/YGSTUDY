    JavaScript 비트 연산자 개념과 사용법¶

    배경¶

    비트 연산자는 숫자를 32비트 정수로 변환하여 비트 단위로 연산을 수행하는 연산자입니다. JavaScript에서는 다양한 비트 연산자를 제공하여 효율적인 비트 조작과 플래그 처리를 할 수 있습니다.

    비트 연산자의 필요성¶

    • 플래그 처리: 여러 상태를 하나의 숫자로 표현
    • 성능 최적화: 빠른 비트 단위 연산
    • 메모리 효율성: 적은 메모리로 많은 정보 저장
    • 암호화/압축: 비트 단위 데이터 처리

    기본 개념¶

    • 32비트 정수: 모든 비트 연산은 32비트 정수로 변환
    • 비트 단위 연산: 각 비트를 개별적으로 처리
    • 2의 보수: 음수 표현 방식
    • 시프트 연산: 비트를 좌우로 이동

    핵심¶

    1. 비트 AND (&)¶

    기본 동작¶

    // 두 숫자의 각 비트를 비교하여 둘 다 1인 경우에만 1 반환
const a = 5;  // 0101 (2진수)
const b = 3;  // 0011 (2진수)
const result = a & b;  // 0001 (1)

// 비트별 연산 과정
// 0101 (5)
// 0011 (3)
// ----
// 0001 (1) - 결과

    실제 활용¶

    // 플래그 확인
const FLAG_A = 1;  // 0001
const FLAG_B = 2;  // 0010
const FLAG_C = 4;  // 0100

let flags = FLAG_A | FLAG_C;  // 0101 (플래그 A와 C 설정)
console.log(flags & FLAG_A);  // 1 (플래그 A가 설정됨)
console.log(flags & FLAG_B);  // 0 (플래그 B가 설정되지 않음)

// 짝수/홀수 판별
function isEven(num) {
    return (num & 1) === 0;
}

console.log(isEven(4));  // true
console.log(isEven(7));  // false

// 마지막 비트 제거
function removeLastBit(num) {
    return num & (num - 1);
}

console.log(removeLastBit(12));  // 8 (1100 -> 1000)

    2. 비트 OR (|)¶

    기본 동작¶

    // 두 숫자의 각 비트를 비교하여 하나라도 1이면 1 반환
const a = 5;  // 0101
const b = 3;  // 0011
const result = a | b;  // 0111 (7)

// 비트별 연산 과정
// 0101 (5)
// 0011 (3)
// ----
// 0111 (7) - 결과

    실제 활용¶

    // 플래그 설정
const FLAG_A = 1;  // 0001
const FLAG_B = 2;  // 0010
const FLAG_C = 4;  // 0100

let flags = 0;
flags |= FLAG_A;  // 플래그 A 설정
flags |= FLAG_C;  // 플래그 C 설정
console.log(flags);  // 5 (0101)

// 기본값 설정
function setDefaultValue(value, defaultValue) {
    return value | defaultValue;
}

console.log(setDefaultValue(0, 10));  // 10
console.log(setDefaultValue(5, 10));  // 15 (5 | 10 = 15)

    setDefaultValue 라는 이름은 이 함수가 하는 일과 맞지 않는다. 값이 0 일 때만 우연히 기본값처럼 보이고, 값이 있으면 두 수를 비트로 합쳐 버린다. 5 를 넣었는데 15 가 나오는 것은 기본값 처리가 아니다. 기본값이 목적이면 value ?? defaultValue 를 쓴다. 비트 OR 은 플래그를 켜는 연산이지 값 선택 연산이 아니다.

    3. 비트 XOR (^)¶

    기본 동작¶

    // 두 숫자의 각 비트를 비교하여 서로 다른 경우에만 1 반환
const a = 5;  // 0101
const b = 3;  // 0011
const result = a ^ b;  // 0110 (6)

// 비트별 연산 과정
// 0101 (5)
// 0011 (3)
// ----
// 0110 (6) - 결과

    실제 활용¶

    // 플래그 토글
const FLAG_A = 1;
const FLAG_B = 2;

let flags = FLAG_A;  // 0001
flags ^= FLAG_A;     // 플래그 A 토글 (제거)
flags ^= FLAG_B;     // 플래그 B 토글 (추가)
console.log(flags);  // 2 (0010)

// 값 교환 (XOR 스왑)
let x = 5, y = 10;
x = x ^ y;
y = x ^ y;
x = x ^ y;
console.log(x, y);  // 10, 5

// 중복 제거
function findSingleNumber(nums) {
    let result = 0;
    for (const num of nums) {
        result ^= num;
    }
    return result;
}

console.log(findSingleNumber([4, 1, 2, 1, 2]));  // 4

    XOR 스왑은 면접 단골이지만 실무에서 쓰면 안 된다. 비트 연산이 피연산자를 32비트 정수로 잘라내기 때문에 정수가 아니거나 범위를 벗어나면 값이 바뀐다.

    let x = 1.5, y = 2.5;
x = x ^ y; y = x ^ y; x = x ^ y;
console.log(x, y);        // 2 1   ← 소수점이 사라졌다

let p = 2 ** 31, q = 1;
p = p ^ q; q = p ^ q; p = p ^ q;
console.log(p, q);        // 1 -2147483648   ← 부호가 뒤집혔다

    같은 값을 두 번 XOR 하면 원래대로 돌아온다는 성질은 32비트 안에서만 성립한다. 자바스크립트의 수는 배정밀도 부동소수점이라 2^31 이상, 소수점, NaN 이 모두 이 전제를 깬다. [a, b] = [b, a] 를 쓰면 이런 조건이 없다.

    같은 이유로 findSingleNumber 도 배열에 2^31 이상의 값이나 소수가 섞이면 틀린 답을 낸다.

    4. 비트 NOT (~)¶

    기본 동작¶

    // 모든 비트를 반전 (0은 1로, 1은 0으로)
const a = 5;  // 0101
const result = ~a;  // -6

// 비트별 연산 과정
// ~0101 = 1010 (2의 보수)
// 32비트: 11111111111111111111111111111010

    실제 활용¶

    // 플래그 제거
const FLAG_A = 1;
const FLAG_B = 2;
const FLAG_C = 4;

let flags = FLAG_A | FLAG_B | FLAG_C;  // 7 (0111)
flags &= ~FLAG_B;  // 플래그 B 제거
console.log(flags);  // 5 (0101)

// 음수 변환
function negate(num) {
    return ~num + 1;
}

console.log(negate(5));  // -5
console.log(negate(-3)); // 3

// 범위 확인
function isInRange(num, min, max) {
    return (num >= min) & (num <= max);
}

console.log(isInRange(5, 1, 10));  // 1 (true)
console.log(isInRange(15, 1, 10)); // 0 (false)

    5. 시프트 연산자¶

    왼쪽 시프트 (<<)¶

    // 모든 비트를 왼쪽으로 이동, 오른쪽은 0으로 채움
const a = 5;  // 0101
const result = a << 2;  // 10100 (20)

// 비트별 연산 과정
// 0101 -> 10100 (2비트 왼쪽 이동)

// 2의 거듭제곱 계산
function powerOfTwo(n) {
    return 1 << n;
}

console.log(powerOfTwo(3));  // 8 (2^3)
console.log(powerOfTwo(5));  // 32 (2^5)

    powerOfTwo 는 n 이 31 이 되는 순간 무너진다.

    1 << 30;   //  1073741824   맞다
1 << 31;   // -2147483648   음수가 나온다
1 << 32;   //  1            0으로 돌아가지 않고 1<<0 이 된다
1 << 33;   //  2

    시프트 결과는 부호 있는 32비트 정수라 31번째 비트가 부호 비트다. 그리고 시프트 횟수는 하위 5비트만 쓰기 때문에 << 32 는 << 0 과 같다. 0 이 되지도, 에러가 나지도 않고 조용히 처음으로 되돌아온다.

    플래그를 32개 넘게 정의하다 이걸 만난다. 33번째 플래그가 1번째 플래그와 같은 값이 되어 권한 두 개가 하나로 붙는 식이다. 플래그가 31개를 넘어갈 것 같으면 1n << 33n 처럼 BigInt 를 쓰거나 아예 Set 으로 바꾼다.

    2 ** n 은 이 제약이 없다. 비트 시프트로 거듭제곱을 계산하는 것은 지금은 가독성만 해친다.

    오른쪽 시프트 (>>)¶

    // 모든 비트를 오른쪽으로 이동, 왼쪽은 부호 비트로 채움
const a = 20;  // 10100
const result = a >> 2;  // 00101 (5)

// 비트별 연산 과정
// 10100 -> 00101 (2비트 오른쪽 이동)

// 2로 나누기
function divideByTwo(num) {
    return num >> 1;
}

console.log(divideByTwo(10));  // 5
console.log(divideByTwo(7));   // 3 (정수 나눗셈)

    "정수 나눗셈"이라는 설명이 음수에서 어긋난다. >> 는 버림이 아니라 내림이다.

    -7 >> 1;              // -4
Math.floor(-7 / 2);   // -4   ← 이쪽과 같다
(-7 / 2) | 0;         // -3   ← 0 쪽으로 자르면 이 값
Math.trunc(-7 / 2);   // -3

    양수만 넣어 테스트하면 둘이 같아 보여서 통과한다. 좌표나 차이값처럼 음수가 들어올 수 있는 자리에서 1 씩 어긋나기 시작한다. 어느 쪽을 원하는지 정한 다음 Math.floor 나 Math.trunc 를 이름 그대로 쓰는 편이 안전하다.

    | 0 과 >> 0 도 서로 다르지 않다 — 둘 다 0 쪽으로 자르는 32비트 변환이다. 5.9 | 0 은 5, -5.9 | 0 은 -5 다.

    부호 없는 오른쪽 시프트 (>>>)¶

    // 모든 비트를 오른쪽으로 이동, 왼쪽은 항상 0으로 채움
const a = -20;  // 11111111111111111111111111101100
const result = a >>> 2;  // 00111111111111111111111111111011

// 양수로 변환
function toUnsigned(num) {
    return num >>> 0;
}

console.log(toUnsigned(-1));  // 4294967295

    >>> 0 을 "정수로 만드는 안전한 방법"으로 쓰는 코드가 많은데, 32비트를 넘어가면 조용히 감싼다.

    4294967296 >>> 0;   // 0   (2^32)
4294967297 >>> 0;   // 1
1e21 | 0;           // -559939584

    에러도, NaN 도 아니고 그럴듯한 숫자가 나온다. 숫자가 아닌 값이 들어와도 마찬가지다.

    NaN | 0;        // 0
Infinity | 0;   // 0
'12abc' | 0;    // 0
undefined | 0;  // 0

    Number('12abc') 는 NaN 이라 검사에 걸리지만 '12abc' | 0 은 0 이라 그냥 통과한다. 입력 검증에 비트 연산을 쓰면 잘못된 값을 정상값으로 바꿔서 통과시키는 셈이 된다. 검증은 Number.isInteger 로 하고, 비트 연산은 이미 정수인 것이 확실한 자리에만 쓴다.

    예시¶

    1. 실제 사용 사례¶

    권한 시스템¶

    // 권한 플래그 정의
const PERMISSIONS = {
    READ: 1,      // 0001
    WRITE: 2,     // 0010
    DELETE: 4,    // 0100
    ADMIN: 8      // 1000
};

class User {
    constructor(name) {
        this.name = name;
        this.permissions = 0;
    }

    // 권한 부여
    grantPermission(permission) {
        this.permissions |= permission;
    }

    // 권한 제거
    revokePermission(permission) {
        this.permissions &= ~permission;
    }

    // 권한 확인
    hasPermission(permission) {
        return (this.permissions & permission) === permission;
    }

    // 모든 권한 확인
    hasAllPermissions(...permissions) {
        const required = permissions.reduce((acc, perm) => acc | perm, 0);
        return (this.permissions & required) === required;
    }

    // 권한 목록 반환
    getPermissionList() {
        const list = [];
        for (const [name, flag] of Object.entries(PERMISSIONS)) {
            if (this.hasPermission(flag)) {
                list.push(name);
            }
        }
        return list;
    }
}

// 사용 예시
const user = new User('Alice');
user.grantPermission(PERMISSIONS.READ);
user.grantPermission(PERMISSIONS.WRITE);

console.log(user.hasPermission(PERMISSIONS.READ));  // true
console.log(user.hasPermission(PERMISSIONS.DELETE)); // false
console.log(user.getPermissionList()); // ['READ', 'WRITE']

    hasPermission 의 괄호는 장식이 아니다. & 는 === 보다 우선순위가 낮다. 괄호를 빼면 비교가 먼저 일어난다.

    const flags = FLAG_A | FLAG_C;          // 5 — B 는 꺼져 있다

flags & FLAG_B === FLAG_B               // 1     → if 에서 true
(flags & FLAG_B) === FLAG_B             // false → 올바른 답

    앞의 식은 flags & (FLAG_B === FLAG_B), 즉 5 & 1 로 계산된다. 없는 권한을 있다고 판정한다. 권한 검사에서 이게 나면 곧바로 보안 결함이다. 게다가 FLAG_A 로 테스트하면 우연히 맞는 답이 나와서 테스트를 통과한다.

    C 계열 언어에서 넘어온 이 우선순위는 지금 기준으로 보면 사고에 가깝다. 비트 연산과 비교를 한 식에 섞을 때는 우선순위를 외우려 하지 말고 무조건 괄호를 친다.

    hasPermission 을 (this.permissions & permission) !== 0 이 아니라 === permission 으로 쓴 것도 의도가 있다. 인자로 READ | WRITE 처럼 여러 비트를 한꺼번에 넘겼을 때 !== 0 은 하나만 있어도 통과하지만 === permission 은 전부 있어야 통과한다.

    색상 처리¶

    // RGB 색상을 하나의 정수로 표현
function rgbToInt(r, g, b) {
    return (r << 16) | (g << 8) | b;
}

function intToRgb(color) {
    return {
        r: (color >> 16) & 255,
        g: (color >> 8) & 255,
        b: color & 255
    };
}

// 사용 예시
const red = rgbToInt(255, 0, 0);
const green = rgbToInt(0, 255, 0);
const blue = rgbToInt(0, 0, 255);

console.log(red);   // 16711680
console.log(green); // 65280
console.log(blue);  // 255

const color = intToRgb(16711680);
console.log(color); // { r: 255, g: 0, b: 0 }

    2. 고급 패턴¶

    비트 마스킹¶

    // 특정 비트 패턴 추출
function extractBits(num, start, length) {
    const mask = (1 << length) - 1;
    return (num >> start) & mask;
}

// 비트 설정
function setBits(num, start, length, value) {
    const mask = (1 << length) - 1;
    const cleared = num & ~(mask << start);
    return cleared | (value << start);
}

// 사용 예시
const data = 0b10110110; // 182
console.log(extractBits(data, 2, 3)); // 6 (110)
console.log(setBits(data, 2, 3, 7));  // 190 (10111110)

    비트 카운팅¶

    // 1의 개수 세기 (Brian Kernighan 알고리즘)
function countSetBits(num) {
    let count = 0;
    while (num) {
        num &= (num - 1);
        count++;
    }
    return count;
}

// 최하위 비트 찾기
function getLowestSetBit(num) {
    return num & -num;
}

// 최하위 비트 제거
function removeLowestSetBit(num) {
    return num & (num - 1);
}

// 사용 예시
console.log(countSetBits(15));  // 4 (1111)
console.log(getLowestSetBit(12)); // 4 (1100 -> 0100)
console.log(removeLowestSetBit(12)); // 8 (1100 -> 1000)

    운영 팁¶

    성능 최적화¶

    빠른 연산¶

    // 2의 거듭제곱 확인
function isPowerOfTwo(num) {
    return num > 0 && (num & (num - 1)) === 0;
}

// 2로 나누기 (정수)
function fastDivideByTwo(num) {
    return num >> 1;
}

// 2 곱하기
function fastMultiplyByTwo(num) {
    return num << 1;
}

// 나머지 연산 (2의 거듭제곱으로 나누기)
function fastModulo(num, divisor) {
    return num & (divisor - 1);
}

// 사용 예시
console.log(isPowerOfTwo(8));  // true
console.log(isPowerOfTwo(10)); // false
console.log(fastModulo(17, 8)); // 1 (17 % 8)

    fastModulo 는 음수에서 % 와 다른 답을 낸다.

    -17 & 7;      //  7
-17 % 8;      // -1

    자바스크립트의 % 는 나머지 부호를 왼쪽 피연산자에 맞추지만 & 는 언제나 양수를 낸다. 해시 값을 버킷 개수로 나눌 때처럼 "항상 0 이상"이 필요한 자리에서는 오히려 & 쪽이 맞다. 문제는 함수 이름이 fastModulo 라 % 의 빠른 대체품처럼 읽힌다는 점이다. 둘은 다른 연산이다.

    이름에 붙은 "fast" 도 그대로 믿을 말은 아니다. 이 문서 어디에도 잰 값이 없다. 비트 트릭을 고를 이유는 속도가 아니라 하려는 일이 실제로 비트 조작일 때다.

    에러 처리¶

    안전한 비트 연산¶

    // 32비트 정수 범위 확인
function safeBitwiseOperation(num) {
    // 32비트 정수로 변환
    return num >>> 0;
}

// 범위 검증
function validateBitRange(num, bits) {
    if (bits < 1 || bits > 32) {
        throw new Error('비트 수는 1-32 범위여야 합니다.');
    }

    const maxValue = (1 << bits) - 1;
    if (num < 0 || num > maxValue) {
        throw new Error(`값은 0-${maxValue} 범위여야 합니다.`);
    }

    return num;
}

// 사용 예시
try {
    const safeNum = safeBitwiseOperation(123456789);
    const validated = validateBitRange(255, 8);
    console.log('안전한 연산 완료');
} catch (error) {
    console.error('오류:', error.message);
}

    validateBitRange 는 "1-32 범위"를 허용한다고 써 놓고 정작 위쪽 끝에서 부서진다. maxValue 를 (1 << bits) - 1 로 구하기 때문이다.

    validateBitRange(255, 32);
// Error: 값은 0-0 범위여야 합니다.      ← (1<<32)-1 === 0

validateBitRange(255, 31);
// Error: 값은 0--2147483649 범위여야 합니다.   ← (1<<31)-1 이 음수

    에러 메시지의 0--2147483649 를 보고 나서야 계산이 깨진 걸 알게 된다. bits 가 31 이상이면 어떤 값을 넣어도 통과하지 못한다. 32비트 전체를 다루려고 만든 검증 함수가 32비트에서만 못 쓰는 셈이다.

    2 ** bits - 1 로 바꾸면 시프트를 거치지 않아 bits === 32 에서도 4294967295 가 나온다. 32비트를 다루는 코드에서 상한 자체를 32비트 연산으로 계산하면 안 된다 — 상한은 언제나 표현 범위 밖에 있다.

    바로 위 safeBitwiseOperation 도 이름과 달리 안전하지 않다. >>> 0 은 범위를 벗어난 값을 거르는 게 아니라 감싸서 다른 값으로 만든다.
