## TDD (Test-Driven Development) 방법론

### 1. TDD 핵심 개념

#### TDD란?
TDD(Test-Driven Development, 테스트 주도 개발)는 테스트를 먼저 작성하고, 이후에 실제 코드를 구현하는 방식입니다. 즉, 기능을 개발하기 전에 먼저 테스트를 작성한 후, 테스트를 통과할 수 있도록 코드를 작성하는 개발 방법론입니다.

#### TDD의 핵심 원칙
- **테스트 우선**: 기능 구현 전에 테스트를 먼저 작성
- **작은 단위**: 작은 기능 단위로 테스트와 구현을 반복
- **지속적 리팩토링**: 코드 개선을 통한 품질 향상

#### TDD의 장점
- 코드의 품질과 신뢰성 향상
- 설계 개선 및 리팩토링 용이
- 문서화 효과
- 버그 조기 발견

### 2. TDD 개발 프로세스

TDD는 **Red → Green → Refactor** 3단계를 반복합니다.

#### Red (실패) 단계
```javascript
// 1. 먼저 테스트를 작성하고 실행 → 당연히 처음에는 실패
describe('UserService', () => {
  describe('createUser', () => {
    it('should create a new user with valid data', () => {
      // Given
      const userData = {
        email: 'test@example.com',
        password: 'password123',
        name: 'Test User'
      };

      // When
      const result = UserService.createUser(userData);

      // Then
      expect(result).toBeDefined();
      expect(result.email).toBe(userData.email);
      expect(result.name).toBe(userData.name);
      expect(result.id).toBeDefined();
    });
  });
});
```

#### Green (성공) 단계
```javascript
// 2. 테스트를 통과할 수 있는 최소한의 코드 작성
class UserService {
  static createUser(userData) {
    // 최소한의 구현으로 테스트 통과
    return {
      id: Date.now().toString(),
      email: userData.email,
      name: userData.name,
      createdAt: new Date()
    };
  }
}
```

#### Refactor (리팩토링) 단계
```javascript
// 3. 코드를 개선하고 최적화
class UserService {
  static createUser(userData) {
    // 입력 검증 추가
    this.validateUserData(userData);
    
    // 비즈니스 로직 개선
    const hashedPassword = this.hashPassword(userData.password);
    
    // 데이터베이스 저장
    const user = {
      id: this.generateUserId(),
      email: userData.email,
      name: userData.name,
      password: hashedPassword,
      createdAt: new Date()
    };
    
    return this.saveUser(user);
  }

  static validateUserData(userData) {
    if (!userData.email || !userData.password || !userData.name) {
      throw new Error('필수 필드가 누락되었습니다.');
    }
    
    if (!this.isValidEmail(userData.email)) {
      throw new Error('유효하지 않은 이메일 형식입니다.');
    }
  }

  static hashPassword(password) {
    const crypto = require('crypto');
    return crypto.createHash('sha256').update(password).digest('hex');
  }

  static generateUserId() {
    return `user_${Date.now()}_${Math.random().toString(36).substr(2, 9)}`;
  }

  static saveUser(user) {
    // 실제 데이터베이스 저장 로직
    return user;
  }
}
```

### 3. TDD 실전 예제: 주문 시스템

#### 단계 1: 주문 생성 테스트 작성
```javascript
// tests/order.test.js
describe('OrderService', () => {
  describe('createOrder', () => {
    it('should create order with valid items', () => {
      // Given
      const orderData = {
        userId: 'user123',
        items: [
          { productId: 'prod1', quantity: 2, price: 1000 },
          { productId: 'prod2', quantity: 1, price: 2000 }
        ]
      };

      // When
      const order = OrderService.createOrder(orderData);

      // Then
      expect(order).toBeDefined();
      expect(order.id).toBeDefined();
      expect(order.userId).toBe(orderData.userId);
      expect(order.items).toHaveLength(2);
      expect(order.totalAmount).toBe(4000); // (2 * 1000) + (1 * 2000)
      expect(order.status).toBe('PENDING');
    });

    it('should throw error for empty items', () => {
      // Given
      const orderData = {
        userId: 'user123',
        items: []
      };

      // When & Then
      expect(() => OrderService.createOrder(orderData))
        .toThrow('주문 항목이 없습니다.');
    });

    it('should calculate total amount correctly', () => {
      // Given
      const orderData = {
        userId: 'user123',
        items: [
          { productId: 'prod1', quantity: 3, price: 500 },
          { productId: 'prod2', quantity: 2, price: 1500 }
        ]
      };

      // When
      const order = OrderService.createOrder(orderData);

      // Then
      expect(order.totalAmount).toBe(4500); // (3 * 500) + (2 * 1500)
    });
  });
});
```

#### 단계 2: 최소 구현으로 테스트 통과
```javascript
// src/services/OrderService.js
class OrderService {
  static createOrder(orderData) {
    if (!orderData.items || orderData.items.length === 0) {
      throw new Error('주문 항목이 없습니다.');
    }

    const totalAmount = orderData.items.reduce((sum, item) => {
      return sum + (item.quantity * item.price);
    }, 0);

    return {
      id: `order_${Date.now()}`,
      userId: orderData.userId,
      items: orderData.items,
      totalAmount: totalAmount,
      status: 'PENDING',
      createdAt: new Date()
    };
  }
}

module.exports = OrderService;
```

#### 단계 3: 리팩토링 및 개선
```javascript
// src/services/OrderService.js
const { v4: uuidv4 } = require('uuid');
const OrderRepository = require('../repositories/OrderRepository');
const ProductService = require('./ProductService');

class OrderService {
  static async createOrder(orderData) {
    // 입력 검증
    this.validateOrderData(orderData);
    
    // 상품 재고 확인
    await this.validateProductAvailability(orderData.items);
    
    // 주문 생성
    const order = this.buildOrder(orderData);
    
    // 데이터베이스 저장
    const savedOrder = await OrderRepository.save(order);
    
    // 이벤트 발행
    await this.publishOrderCreatedEvent(savedOrder);
    
    return savedOrder;
  }

  static validateOrderData(orderData) {
    if (!orderData.userId) {
      throw new Error('사용자 ID가 필요합니다.');
    }
    
    if (!orderData.items || orderData.items.length === 0) {
      throw new Error('주문 항목이 없습니다.');
    }
    
    orderData.items.forEach((item, index) => {
      if (!item.productId || !item.quantity || !item.price) {
        throw new Error(`주문 항목 ${index + 1}의 필수 정보가 누락되었습니다.`);
      }
      
      if (item.quantity <= 0) {
        throw new Error(`주문 항목 ${index + 1}의 수량은 0보다 커야 합니다.`);
      }
      
      if (item.price <= 0) {
        throw new Error(`주문 항목 ${index + 1}의 가격은 0보다 커야 합니다.`);
      }
    });
  }

  static async validateProductAvailability(items) {
    for (const item of items) {
      const product = await ProductService.getProduct(item.productId);
      
      if (!product) {
        throw new Error(`상품 ${item.productId}를 찾을 수 없습니다.`);
      }
      
      if (product.stock < item.quantity) {
        throw new Error(`상품 ${product.name}의 재고가 부족합니다. (요청: ${item.quantity}, 재고: ${product.stock})`);
      }
    }
  }

  static buildOrder(orderData) {
    const totalAmount = orderData.items.reduce((sum, item) => {
      return sum + (item.quantity * item.price);
    }, 0);

    return {
      id: uuidv4(),
      userId: orderData.userId,
      items: orderData.items.map(item => ({
        productId: item.productId,
        quantity: item.quantity,
        price: item.price,
        subtotal: item.quantity * item.price
      })),
      totalAmount: totalAmount,
      status: 'PENDING',
      createdAt: new Date(),
      updatedAt: new Date()
    };
  }

  static async publishOrderCreatedEvent(order) {
    const EventBus = require('../events/EventBus');
    await EventBus.publish('order.created', {
      orderId: order.id,
      userId: order.userId,
      totalAmount: order.totalAmount,
      timestamp: order.createdAt
    });
  }
}

module.exports = OrderService;
```

#### 리팩토링 단계에서 **테스트가 따라오지 않았다**

TDD 의 세 번째 단계는 "테스트를 초록불로 유지하면서" 구조를 바꾸는 것인데, 위 리팩토링은 그 조건을 깬다. 단계 2 의 `createOrder` 는 동기였고 단계 3 에서 `async` 가 됐다. 그런데 단계 1 의 테스트는 그대로다.

```javascript
const order = OrderService.createOrder(orderData);   // 이제 Promise 를 받는다
expect(order).toBeDefined();
expect(order.totalAmount).toBe(4000);
```

```
order 의 정체: Promise / order.totalAmount = undefined
```

**`expect(order).toBeDefined()` 는 그대로 통과한다.** Promise 도 정의된 값이기 때문이다. 그 다음 줄에서야 실패한다. 첫 단언이 통과하는 게 함정이라, "일부만 깨졌으니 사소한 문제"로 보이게 만든다.

예외 테스트는 더 조용히 무너진다.

```javascript
expect(() => OrderService.createOrder(orderData)).toThrow('주문 항목이 없습니다.');
```

```
결과: expect(received).toThrow(expected)
```

`async` 함수는 **던지지 않고 거부된 Promise 를 반환한다.** 그래서 `toThrow` 가 잡을 게 없다. 여기서는 실패로 드러났지만, 만약 `expect(...).rejects` 없이 그냥 `OrderService.createOrder(bad)` 를 호출만 해두면 **테스트는 통과하고 처리되지 않은 거부만 남는다.** 비동기로 바꾼 뒤에는 반드시 이 형태로 고친다.

```javascript
const order = await OrderService.createOrder(orderData);
await expect(OrderService.createOrder(bad)).rejects.toThrow('주문 항목이 없습니다.');
```

#### `validateOrderData` 의 수량 검사는 도달하지 않는다

같은 리팩토링에 논리 오류가 하나 더 있다.

```javascript
if (!item.productId || !item.quantity || !item.price) {
  throw new Error(`주문 항목 ${index + 1}의 필수 정보가 누락되었습니다.`);
}
if (item.quantity <= 0) {
  throw new Error(`주문 항목 ${index + 1}의 수량은 0보다 커야 합니다.`);
}
```

`quantity: 0` 은 falsy 라 **첫 번째 조건에 먼저 걸린다.**

```
!item.quantity = true  → "필수 정보 누락" 분기로 간다
item.quantity <= 0 분기는 도달 불가: false
```

`quantity <= 0` 검사는 **음수일 때만** 동작한다. 사용자는 수량을 0 으로 보냈는데 "필수 정보가 누락되었습니다" 라는 엉뚱한 메시지를 받는다. `price: 0`(무료 상품)도 마찬가지로 "누락"으로 처리된다.

값의 존재 여부를 볼 때는 falsy 대신 `undefined`/`null` 을 명시적으로 확인한다.

```javascript
if (item.productId == null || item.quantity == null || item.price == null) { ... }
```

이 부류는 **`0` 과 `''` 과 `false` 가 유효한 값일 수 있는 자리**에서 반복해서 나온다. 재고 0, 가격 0, 할인율 0, 검색어 빈 문자열 전부 같은 함정이다.

### 4. TDD 모범 사례

#### 테스트 작성 원칙
```javascript
// 좋은 예시
describe('UserService', () => {
  describe('authenticateUser', () => {
    it('should return user when valid credentials provided', async () => {
      // Given - 명확한 테스트 데이터 설정
      const credentials = {
        email: 'test@example.com',
        password: 'validPassword123'
      };
      const expectedUser = {
        id: 'user123',
        email: 'test@example.com',
        name: 'Test User'
      };
      
      // Mock 설정
      UserRepository.findByEmail.mockResolvedValue(expectedUser);
      PasswordService.verify.mockResolvedValue(true);

      // When - 단일 동작 실행
      const result = await UserService.authenticateUser(credentials);

      // Then - 명확한 검증
      expect(result).toEqual(expectedUser);
      expect(UserRepository.findByEmail).toHaveBeenCalledWith(credentials.email);
      expect(PasswordService.verify).toHaveBeenCalledWith(
        credentials.password, 
        expectedUser.passwordHash
      );
    });

    it('should throw error when invalid credentials provided', async () => {
      // Given
      const credentials = {
        email: 'test@example.com',
        password: 'wrongPassword'
      };
      
      UserRepository.findByEmail.mockResolvedValue(null);

      // When & Then
      await expect(UserService.authenticateUser(credentials))
        .rejects
        .toThrow('인증에 실패했습니다.');
    });
  });
});
```

## 테스트 피라미드 실제 구현

### 1. 테스트 피라미드 개념 및 구조

#### 테스트 피라미드 구조
```
        E2E Tests (10%)
       ┌─────────────────┐
      │   Playwright     │
      │   사용자 시나리오  │
      └─────────────────┘
     ┌─────────────────────┐
    │   Integration Tests  │
    │   Supertest API      │
    │   (20%)              │
    └─────────────────────┘
   ┌─────────────────────────┐
  │     Unit Tests (70%)     │
  │     Jest 순수 함수       │
  │     비즈니스 로직        │
  └─────────────────────────┘
```

#### 70 / 20 / 10 은 목표치가 아니다

이 비율은 "이렇게 나오는 경우가 많더라"는 관찰이지 지켜야 할 기준이 아니다. 숫자를 목표로 삼으면 두 방향으로 틀어진다.

- **비율을 맞추려고 단위 테스트를 늘린다.** getter·setter·단순 위임 함수에 테스트를 붙이면 개수는 늘지만 잡히는 버그는 없다. 대신 리팩터링할 때마다 고쳐야 할 테스트만 늘어난다.
- **E2E 가 10% 를 넘으면 안 된다고 믿는다.** 결제·로그인처럼 실패 비용이 큰 흐름은 개수와 무관하게 끝에서 끝까지 확인해야 한다.

피라미드가 실제로 말하는 것은 개수가 아니라 **되먹임 속도와 원인 지목 능력**이다.

| | 단위 | 통합 | E2E |
|---|---|---|---|
| 실패했을 때 알려 주는 것 | 어느 함수가 틀렸는지 | 어느 경계가 안 맞는지 | 사용자가 못 쓴다는 사실 |
| 실패 원인 찾는 비용 | 낮다 | 중간 | 높다 — 어디가 원인인지 모른다 |
| 깨지는 빈도 | 리팩터링할 때 | 계약이 바뀔 때 | 환경·타이밍·네트워크에도 깨진다 |
| 없을 때 놓치는 것 | 경계값·분기 | 직렬화·DB 제약·인증 | "전부 통과했는데 화면이 안 뜬다" |

**위로 갈수록 하나당 비용이 커지므로 적게 두는 것**이지, 위쪽이 덜 중요해서가 아니다. 그래서 개수를 세는 대신 이렇게 묻는 편이 낫다 — 이 테스트가 실패했을 때 **어디를 고쳐야 할지 바로 아는가.** 모르겠다면 그 테스트는 한 층 아래에 있어야 한다.

단위 테스트에서 특히 조심할 것은 **모킹의 양**이다. 의존성을 전부 목으로 바꾸면 그 테스트는 "내가 목을 이렇게 설정했다"를 확인할 뿐이고, 실제 구현이 바뀌어도 통과한다. 목이 셋을 넘어가기 시작하면 그건 단위 테스트가 필요 없다는 신호가 아니라 **그 함수가 너무 많은 것을 알고 있다는 신호**인 경우가 많다.

#### 이 테스트 파일은 그대로 두면 첫 실행에서 멈춘다

import 목록과 본문을 대조하면 세 함수가 정의 없이 쓰인다.

| 쓰인 함수 | 어디서 왔나 |
|---|---|
| `getAuthToken` | 없음 |
| `cleanupTestData` | 없음 |
| `createTestOrder` | 없음 |

`beforeAll` 안의 `getAuthToken(testUser)` 에서 `ReferenceError` 가 나고, 그러면 그 아래 `describe` 블록 전체가 실행되지 않은 채 실패한다. 통합 테스트에서 실제로 자주 겪는 형태이기도 하다 — **헬퍼가 늘어나는 속도가 import 를 정리하는 속도보다 빠르다.** 헬퍼를 파일 하나에 모아 한 줄로 가져오거나, `setupFilesAfterEach` 로 전역에 심어 두면 이 부류가 줄어든다.

#### `beforeAll` 로 만든 픽스처와 `beforeEach` 정리가 부딪힌다

```javascript
beforeAll(async () => {
  testUser = await createTestUser();      // 한 번만 만든다
  testProduct = await createTestProduct();
});

beforeEach(async () => {
  await cleanupTestData();                 // 매 테스트 전에 지운다
});
```

`cleanupTestData()` 가 테이블을 비우는 종류라면 **첫 테스트가 끝난 시점에 `testUser` 와 `testProduct` 가 DB 에서 사라진다.** 변수에는 객체가 남아 있으니 두 번째 테스트는 존재하지 않는 사용자 ID 로 요청을 보내고, 외래 키 위반이나 404 로 실패한다. 실패 메시지는 "주문 생성 실패"라 원인이 정리 훅이라는 게 안 보인다.

정리 범위와 픽스처 수명은 같이 정해야 한다. 선택지는 셋이다.

| 방법 | 대가 |
|---|---|
| 픽스처도 `beforeEach` 에서 매번 생성 | 느려진다. 테스트 수 × 생성 비용 |
| 정리 대상에서 픽스처 테이블 제외 | 어느 테이블이 예외인지 계속 기억해야 한다 |
| 각 테스트를 트랜잭션으로 감싸고 롤백 | 가장 깔끔하지만 커밋을 검증하는 테스트에는 못 쓴다 |

여기에 **병렬 실행이 겹치면 더 복잡해진다.** 이 문서의 Jest 설정은 `maxWorkers: '50%'` 이고, Jest 는 테스트 **파일**을 워커에 나눠 동시에 돌린다. 통합 테스트 파일이 두 개 이상이고 모두 같은 DB 를 보면, 한쪽의 `cleanupTestData()` 가 다른 쪽이 방금 만든 데이터를 지운다. 증상은 **가끔 실패하는 테스트**로 나타나고, 다시 돌리면 통과하니 "재실행하면 되네"로 넘어가다 신뢰를 잃는다. 통합 테스트만 `--runInBand` 로 돌리거나, 워커마다 다른 스키마·DB 를 쓰게 해야 한다.

에러 **문구**에 단언을 걸면 문구를 다듬는 것이 테스트를 깨는 일이 된다. 오탈자 수정, 다국어 도입, 문장 정리 — 전부 코드 동작과 무관한 변경인데 CI 가 빨갛게 된다. 몇 번 겪으면 사람들은 문구 고치기를 피하게 되고, 그러면 테스트가 제품을 나쁘게 만든다. 안정적인 식별자에 걸어야 한다.

```javascript

    await page.click('[data-testid="place-order-button"]');

    // Then - 에러 메시지 표시
    await expect(page.locator('[data-testid="payment-error"]')).toBeVisible();
    await expect(page.locator('[data-testid="payment-error"]')).toContainText('유효하지 않은 카드 번호');
  });

  test('should handle out of stock scenario', async ({ page }) => {
    // Given - 재고가 없는 상품
    await page.goto('/products/out-of-stock-product');

    // When - 장바구니에 추가 시도
    await page.click('[data-testid="add-to-cart"]');

    // Then - 재고 부족 메시지 표시
    await expect(page.locator('[data-testid="stock-error"]')).toBeVisible();
    await expect(page.locator('[data-testid="stock-error"]')).toContainText('재고가 부족합니다');
  });
});
```

#### 이 세 테스트는 서로 간섭한다

세 테스트가 **같은 계정 하나**(`test@example.com`)를 쓰고 장바구니를 비우지 않는다. 첫 번째 테스트는 이렇게 단언한다.

```javascript
await expect(page.locator('[data-testid="cart-count"]')).toHaveText('1');
```

두 번째 테스트도 장바구니에 상품을 담는다. 장바구니가 서버에 사용자 단위로 저장된다면 **먼저 돈 테스트가 남긴 항목 때문에 `cart-count` 가 '2' 가 된다.** Playwright 는 기본적으로 파일을 병렬로 돌리므로 순서도 보장되지 않는다. 증상은 "혼자 돌리면 통과, 전체 돌리면 실패"이고, 이게 E2E 를 못 믿게 만드는 가장 흔한 형태다.

**E2E 는 자기가 쓸 데이터를 스스로 만들고 스스로 치워야 한다.** 계정을 테스트마다 새로 만들거나(가장 확실하다), 최소한 워커별로 다른 계정을 쓴다.

```javascript
test.beforeEach(async ({ page }, testInfo) => {
  const email = `e2e-${testInfo.workerIndex}-${Date.now()}@example.com`;
  // 이 계정으로 가입 → 로그인 → 테스트 종료 후 정리
});
```

같은 이유로 `/products/out-of-stock-product` 도 위험하다. **어딘가에 그런 상품이 미리 있어야 한다**는 전제인데, 누군가 재고를 채우거나 상품을 지우면 테스트가 깨진다. 그리고 그 실패는 코드 결함이 아니라 데이터 결함이라 원인을 찾는 데 오래 걸린다. 재고 0 인 상품을 테스트 안에서 만들어 쓰는 편이 낫다.

`:first-child` 선택자도 같은 부류다. 상품 목록의 정렬이 바뀌거나 신규 상품이 등록되면 다른 상품이 담긴다. 무엇이 담겼는지 단언하지 않으므로 **엉뚱한 상품으로 주문이 끝나도 테스트는 통과한다.**

#### `beforeEach` 마다 UI 로 로그인하지 않는다

로그인 폼을 채우는 왕복이 테스트 수만큼 반복된다. 두 가지가 문제다.

- **느리다.** E2E 는 원래 느린데, 본론과 무관한 절차를 매번 붙인다.
- **로그인 UI 가 바뀌면 전부 깨진다.** 로그인 자체를 검증하는 테스트는 하나면 충분한데, 지금은 모든 테스트가 로그인 화면에 의존한다.

Playwright 는 인증 상태를 한 번 만들어 재사용하는 경로를 제공한다. 로그인은 셋업에서 한 번만 하고 나머지 테스트는 그 상태로 시작한다.

```javascript
// auth.setup.js — 한 번만 실행
await page.context().storageState({ path: 'playwright/.auth/user.json' });

// playwright.config.js
use: { storageState: 'playwright/.auth/user.json' }
```

**로그인 흐름을 검증하는 테스트는 따로 하나 남긴다.** 그래야 "로그인이 깨졌다"와 "주문이 깨졌다"가 구분된다.

#### 결제까지 진짜로 실행할 것인가

```javascript
await page.fill('[data-testid="card-number"]', '1234-5678-9012-3456');
await page.click('[data-testid="place-order-button"]');
```

이 테스트가 어디까지 실제로 도달하는지를 문서가 정하지 않았다. 결제 게이트웨이까지 진짜로 간다면 **CI 가 돌 때마다 결제가 발생한다.** 안 간다면 어디서 끊기는지(모의 서버·샌드박스 키·스텁 응답) 명시해야 하고, 그 경계가 곧 "이 E2E 가 실제로 검증하는 범위"다.

E2E 에서 외부 시스템을 어디까지 포함할지는 미리 정해 두는 편이 낫다.

| 대상 | 대개의 선택 |
|---|---|
| 자사 DB·API | 실제로 쓴다 |
| 결제·문자·메일 발송 | 샌드박스나 스텁. 실제 호출은 하지 않는다 |
| 외부 인증(소셜 로그인) | 우회 경로를 만들어 둔다 |
| 서드파티 위젯·광고 | 차단한다. 느리고 자주 깨진다 |

`toContainText('유효하지 않은 카드 번호')` 처럼 화면 문구에 단언하는 것도 앞의 통합 테스트에서 지적한 것과 같다. 다국어를 도입하는 순간 전부 깨진다.

#### ⚠ `moduleNameMapping` 은 존재하지 않는 옵션이다

올바른 이름은 **`moduleNameMapper`** 다. 위 설정 그대로 돌리면 별칭 import 가 전부 깨진다.

```
● Validation Warning:
  Unknown option "moduleNameMapping" with value {"^@/(.*)$": "<rootDir>/src/$1"} was found.
  This is probably a typing mistake.

FAIL t/alias.test.js
  ● Test suite failed to run
    Cannot find module '@/greet' from 't/alias.test.js'
```

(Jest 30.4.1)

경고는 나오지만 **빌드를 멈추지는 않는다.** 그래서 설정 파일이 길어지면 스크롤 위쪽에서 지나가고, 사람들은 `Cannot find module '@/...'` 만 보고 tsconfig 나 webpack alias 를 뒤진다. 정작 원인은 Jest 설정 파일의 오타 한 글자다.

이 부류를 막는 방법은 하나뿐이다 — **경고를 흘려보내지 않는 것.** Jest 는 알 수 없는 옵션을 경고로만 처리하므로, CI 로그에서 `Validation Warning` 을 잡아 실패시키거나 설정을 TypeScript 로 작성해 타입 검사를 받는다.

```javascript
// jest.config.ts — 오타가 컴파일 단계에서 잡힌다
import type { Config } from 'jest';
const config: Config = { moduleNameMapper: { '^@/(.*)$': '<rootDir>/src/$1' } };
export default config;
```

같은 이유로 `collectCoverage: true` 를 설정 파일에 박아두는 것도 다시 생각해 볼 만하다. 커버리지 계측은 모든 실행에 붙기 때문에, 개발 중 `--watch` 로 돌릴 때도 매번 따라온다. `test:coverage` 스크립트에서 `--coverage` 플래그로 켜는 편이 낫다 — 아래 `package.json` 에 이미 그 스크립트가 있으니 설정 쪽 `collectCoverage` 와 중복이기도 하다.

#### 커버리지 80% 가 보장하는 것

```javascript
coverageThreshold: { global: { branches: 80, functions: 80, lines: 80, statements: 80 } }
```

커버리지는 **그 줄이 실행됐는지**만 센다. 실행 결과가 맞는지는 보지 않는다. 그래서 아래 테스트는 커버리지를 100% 로 만들면서 아무것도 검증하지 않는다.

```javascript
it('주문을 생성한다', async () => {
  await orderService.create({ items: [] });   // 단언이 없다
});
```

호출만 하고 `expect` 를 쓰지 않아도 통과하고, 커버리지에는 반영된다. **커버리지가 낮으면 확실히 문제지만, 높다고 안전한 것은 아니다.** 이 비대칭을 알고 써야 숫자에 속지 않는다.

`global` 로 한 덩어리를 거는 것도 다시 볼 만하다. 전체 80% 는 **테스트가 쉬운 파일이 어려운 파일의 빈자리를 메워 주는** 구조다. 유틸리티 함수 수백 줄이 100% 를 찍으면 결제 로직이 20% 여도 전체는 통과한다. 중요한 경로에는 따로 높은 기준을 건다.

```javascript
coverageThreshold: {
  global: { lines: 70 },
  './src/payment/': { lines: 95, branches: 90 },
}
```

네 지표 중 **`branches` 가 가장 정직하다.** `lines` 는 `if (a && b)` 한 줄을 한 번만 밟아도 채워지지만 `branches` 는 조건 조합을 본다. 하나만 본다면 이쪽이다.

그리고 **기존 프로젝트에 임계값을 처음 걸 때는 현재 수치에서 시작한다.** 지금 42% 인데 80% 를 걸면 CI 가 빨간 상태로 굳고, 그러면 사람들은 게이트를 끄거나 무시한다. 현재 값을 바닥으로 잡고 내려가지만 않게 막은 뒤 조금씩 올리는 편이 실제로 작동한다.

#### 실패했을 때 무엇을 할지 미리 정한다

자동화된 테스트에서 가장 비싼 것은 **가끔 실패하는 테스트**다. 재실행하면 통과하니 사람들은 재실행 버튼을 누르기 시작하고, 그 습관이 자리 잡으면 진짜 실패도 함께 무시된다. **테스트 스위트의 가치는 실패했을 때 사람들이 멈추느냐로 결정되고, 그건 신뢰의 문제이지 개수의 문제가 아니다.**

앞에서 본 것들이 전부 그 원인이다 — 공유 픽스처, 병렬 실행 중 DB 간섭, 시간·순서 의존, 외부 시스템 호출. 규칙을 미리 정해 두면 관리가 된다.

| 상황 | 정해 둘 것 |
|---|---|
| 두 번 연속 재실행으로 결과가 갈린다 | 그 테스트를 격리(skip)하고 이슈를 연다. 방치하지 않는다 |
| skip 한 테스트 | 기한을 붙인다. 기한이 지나면 고치거나 지운다 |
| 재시도 설정(`retries`) | E2E 에만 허용하고 단위·통합에는 쓰지 않는다 |
| CI 소요 시간 | 상한을 정한다. 넘으면 테스트를 늘리는 대신 줄인다 |

마지막 줄이 중요하다. **테스트는 자산이자 부채다.** 실행 시간, 유지보수, 거짓 실패가 전부 비용이라, 잡아 주는 버그가 그 비용보다 적은 테스트는 지우는 것이 맞다.

### 2. 테스트 자동화

#### 테스트 자동화 레벨
```javascript
class TestAutomationStrategy {
  constructor() {
    this.automationLevels = {
      level1: '단위 테스트 자동화',
      level2: '통합 테스트 자동화', 
      level3: 'E2E 테스트 자동화',
      level4: '성능 테스트 자동화',
      level5: '보안 테스트 자동화'
    };
  }

  // 테스트 자동화 우선순위 결정
  determinePriority(project) {
    const priorities = [];
    
    // 필수: 단위 테스트와 통합 테스트
    priorities.push({
      level: 1,
      type: 'unit',
      reason: '기본적인 코드 품질 보장',
      effort: 'low',
      impact: 'high'
    });

    priorities.push({
      level: 2, 
      type: 'integration',
      reason: '모듈 간 상호작용 검증',
      effort: 'medium',
      impact: 'high'
    });

    // 선택적: 프로젝트 특성에 따라
    if (project.hasUI) {
      priorities.push({
        level: 3,
        type: 'e2e',

        effort: 'high',
```

```javascript
};
```

#### 이 스크립트가 실제로 재는 것

**"100 사용자"는 초당 100 요청이 아니다.** VU(가상 사용자)는 동시에 도는 실행 흐름의 수이고, 처리량은 한 반복이 얼마나 걸리느냐로 결정된다. 이 스크립트는 엔드포인트 3개를 돌면서 각각 뒤에 `sleep(1)` 을 넣는다.

```
한 VU 의 1회 반복 ≈ 3초 (응답 시간 제외)
VU 100 → 초당 요청 수 ≈ 100 req/s
```

우연히 숫자가 비슷해졌지만 `sleep` 을 빼면 처리량이 몇 배로 뛴다. **부하 목표는 VU 가 아니라 처리량으로 정해야** 다른 사람이 결과를 해석할 수 있다. 목표가 "초당 몇 건"이라면 `constant-arrival-rate` 실행기를 쓰는 편이 정확하다 — VU 를 늘리는 대신 도착률을 고정한다.

**`r.json().length > 0` 은 응답 모양을 가린다.**

```
[1,2,3]                    → true
[]                         → false
{"data":[1,2]}             → false     ← 배열이 아니라서
{"items":[1],"total":1}    → false     ← 페이지네이션 응답
```

목록 API 가 `{ data: [...] }` 형태를 돌려주면 정상 응답인데도 전부 실패로 집계된다. 반대로 응답이 JSON 이 아닐 때(500 에러 HTML 페이지 등)는 `r.json()` 자체가 실패해서 **에러율이 올라가는 게 아니라 스크립트가 그 자리에서 멈춘다.** 부하 테스트에서 가장 알고 싶은 상황이 바로 서버가 5xx 를 뱉기 시작할 때인데, 그때 계측이 죽는 구조다. 상태 코드부터 확인하고 본문 검사는 그 뒤에 둔다.

```javascript
check(response, {
  'status is 200': (r) => r.status === 200,
  'body is list':  (r) => r.status === 200 && Array.isArray(r.json('data') ?? r.json()),
});
```

**임계값이 전부 합산이다.** `http_req_duration: ['p(95)<500']` 은 세 엔드포인트를 뭉쳐서 본다. `/api/users` 가 아무리 느려도 나머지 둘이 빠르면 전체 p95 가 통과한다. 엔드포인트마다 태그를 붙이고 태그별 임계값을 거는 편이 낫다. 그러지 않으면 **"통과했다"가 "어디도 느리지 않다"를 뜻하지 않는다.**

**인증이 없다.** `/api/users` 를 토큰 없이 부르면 둘 중 하나다 — 401 이 나서 아무 부하도 안 걸리거나(인증 미들웨어까지만 재고 있다), 정말로 인증 없이 열려 있거나(그건 부하 문제가 아니라 보안 결함이다). 어느 쪽이든 지금 숫자는 실제 트래픽을 대변하지 못한다. 부하 테스트는 **인증·캐시 워밍·실제 데이터 크기**를 실제와 맞춘 뒤에야 의미가 생긴다. 빈 DB 를 상대로 잰 p95 는 운영 예측에 쓸 수 없다.

마지막으로 이런 종류의 결과는 **비교로만 쓴다.** 같은 스크립트를 같은 환경에서 변경 전후로 돌려 상대 변화를 보는 용도다. 개발 노트북에서 한 번 재고 "우리 서비스는 초당 N건을 견딘다"고 적으면 그 숫자는 근거가 없다.

```javascript
const artillery = require('artillery');
const fs = require('fs');

class ArtilleryManager {
  constructor() {
    this.configFile = 'artillery-config.yml';
    this.resultsDir = './performance-results';
  }

  async runPerformanceTest(testType = 'load') {
    const configs = {
      load: {
        phases: [
          { duration: 60, arrivalRate: 10 },
          { duration: 120, arrivalRate: 20 }
        ],
        ensure: { p95: 500, maxErrorRate: 5 }
      },
      stress: {
        phases: [
          { duration: 60, arrivalRate: 10 },
          { duration: 120, arrivalRate: 50 },
          { duration: 60, arrivalRate: 100 }
        ],
        ensure: { maxErrorRate: 10 }
      },
      spike: {
        phases: [
          { duration: 30, arrivalRate: 10 },
          { duration: 60, arrivalRate: 100 },
          { duration: 30, arrivalRate: 10 }
        ]
      }
    };

    const config = configs[testType];
    const timestamp = new Date().toISOString().replace(/[:.]/g, '-');
    const resultsFile = `${this.resultsDir}/artillery-${testType}-${timestamp}.json`;

    return new Promise((resolve, reject) => {
      const runner = artillery.runner(config, {
        target: process.env.TARGET_URL || 'http://localhost:3000',
        output: resultsFile
      });

      runner.on('done', (results) => {
        console.log(`Performance test completed: ${testType}`);
        resolve({
          type: testType,
          results,
          file: resultsFile
        });
      });

      runner.on('error', (error) => {
        reject(error);
      });
    });
  }

  // 성능 테스트 결과 분석
  analyzeResults(resultsFile) {
    const results = JSON.parse(fs.readFileSync(resultsFile, 'utf8'));
    
    return {
      summary: results.aggregate,
      metrics: {
        requests: results.aggregate.counters['http.requests'],
        errors: results.aggregate.counters['http.request_rate'],
        responseTime: {
          min: results.aggregate.summaries['http.response_time'].min,
          max: results.aggregate.summaries['http.response_time'].max,
          median: results.aggregate.summaries['http.response_time'].median,
          p95: results.aggregate.summaries['http.response_time'].p95,
          p99: results.aggregate.summaries['http.response_time'].p99
        }
      },
      passRate: this.calculatePassRate(results),
      recommendations: this.generateRecommendations(results)
    };
  }
}
```

```javascript
    // 트랜잭션 시작
    await this.beginTransaction();
    
    // 테스트 데이터 초기화
    await this.initializeTestData();
    
    // 정리 작업 등록
    this.registerCleanupTasks();
  }

  // 테스트 종료 시 격리 환경 정리
  async cleanup() {
    try {
      // 등록된 정리 작업 실행
      for (const task of this.cleanupTasks) {
        await task();
      }
      
      // 트랜잭션 롤백
      await this.rollbackTransaction();
      
      console.log('Test isolation cleanup completed');
    } catch (error) {
      console.error('Error during test cleanup:', error);
    }
  }

  async beginTransaction() {
    // 데이터베이스 트랜잭션 시작
    this.transaction = await this.pool.query('BEGIN');
  }

  async rollbackTransaction() {
    // 트랜잭션 롤백으로 모든 변경사항 취소
    await this.pool.query('ROLLBACK');
  }

  // 테스트 데이터 생성 및 추적
  async createTestData(type, data) {
    const record = await this.insertRecord(type, data);
    
    // 생성된 데이터를 추적하여 정리 시 사용
    if (!this.testData.has(type)) {
      this.testData.set(type, []);
    }
    this.testData.get(type).push(record.id);
    
    return record;
  }

  registerCleanupTasks() {
    // 데이터베이스 정리
    this.cleanupTasks.push(async () => {
      for (const [type, ids] of this.testData) {
        await this.deleteRecords(type, ids);
      }
    });

    // 파일 시스템 정리
    this.cleanupTasks.push(async () => {
      await this.cleanupTestFiles();
    });

    // 외부 서비스 정리
    this.cleanupTasks.push(async () => {
      await this.cleanupExternalServices();
    });
  }
}

module.exports = TestIsolationManager;
```

## CI/CD 파이프라인에서의 품질 게이트

### 1. 품질 게이트 정의 및 설정

#### 품질 게이트 매니저
```javascript
// scripts/quality-gate-manager.js
class QualityGateManager {
  constructor() {
    this.gates = {
      unitTests: {
        threshold: 90, // 90% 이상 통과
        weight: 30
      },
      integrationTests: {
        threshold: 85, // 85% 이상 통과
        weight: 25
      },
      codeCoverage: {
        threshold: 80, // 80% 이상 커버리지
        weight: 20
      },
      securityScan: {
        threshold: 0, // 보안 취약점 0개
        weight: 15
      },
      performanceTests: {
        threshold: 500, // 500ms 이하 응답시간
        weight: 10
      }
    };
  }

  async evaluateQuality(allResults) {
    const scores = {};
    let totalScore = 0;
    let totalWeight = 0;

    for (const [gateName, config] of Object.entries(this.gates)) {
      const result = allResults[gateName];
      const score = this.calculateGateScore(gateName, result, config);
      
      scores[gateName] = {
        score,
        passed: score >= config.threshold,
        weight: config.weight
      };
      
      totalScore += score * config.weight;
      totalWeight += config.weight;
    }

    const overallScore = totalWeight > 0 ? totalScore / totalWeight : 0;
    const passed = Object.values(scores).every(gate => gate.passed);

    return {
      overallScore,
      passed,
      gates: scores,
      recommendation: this.generateRecommendation(scores)
    };
  }

  calculateGateScore(gateName, result, config) {
    switch (gateName) {
      case 'unitTests':
        return (result.passed / result.total) * 100;
      
      case 'codeCoverage':
        return result.percentage;
      
      case 'securityScan':
        return result.vulnerabilities === 0 ? 100 : Math.max(0, 100 - result.vulnerabilities * 10);
      
      case 'performanceTests':
        return result.averageResponseTime <= config.threshold ? 100 : 
               Math.max(0, 100 - ((result.averageResponseTime - config.threshold) / config.threshold) * 100);
      
      default:
        return (result.passed / result.total) * 100;
    }
  }
}
```

### 3. 품질 게이트 자동화 스크립트

#### 품질 게이트 평가 스크립트
```javascript
// scripts/evaluate-quality-gate.js
const { execSync } = require('child_process');
const fs = require('fs');
const QualityGateManager = require('./quality-gate-manager');

async function evaluateQualityGate() {
  console.log('Evaluating quality gates...');
  
  const qualityManager = new QualityGateManager();
  const results = {};
  
  try {
    // 1. 단위 테스트 결과 수집
    console.log('Collecting unit test results...');
    results.unitTests = await collectUnitTestResults();
    
    // 2. 통합 테스트 결과 수집
    console.log('Collecting integration test results...');
    results.integrationTests = await collectIntegrationTestResults();
    
    // 3. 코드 커버리지 수집
    console.log('Collecting code coverage...');
    results.codeCoverage = await collectCoverageResults();
    
    // 4. 보안 스캔 결과 수집
    console.log('Collecting security scan results...');
    results.securityScan = await collectSecurityScanResults();
    
    // 5. 성능 테스트 결과 수집
    console.log('Collecting performance test results...');
    results.performanceTests = await collectPerformanceResults();
    
    // 6. 품질 게이트 평가
    console.log('Evaluating quality gates...');
    const evaluation = await qualityManager.evaluateQuality(results);
    
    // 7. 결과 저장
    fs.writeFileSync('quality-report.json', JSON.stringify(evaluation, null, 2));
    
    // 8. 결과 출력
    printQualityReport(evaluation);
    
    // 9. 품질 게이트 통과 여부에 따른 종료 코드 설정
    process.exit(evaluation.passed ? 0 : 1);
    
  } catch (error) {
    console.error('Error evaluating quality gates:', error);
    process.exit(1);
  }
}

async function collectUnitTestResults() {
  try {
    const output = execSync('npm run test:unit -- --json', { encoding: 'utf8' });
    const results = JSON.parse(output);
    
    return {
      total: results.numTotalTests,
      passed: results.numPassedTests,
      failed: results.numFailedTests,
      percentage: (results.numPassedTests / results.numTotalTests) * 100
    };
  } catch (error) {
    console.error('Failed to collect unit test results:', error.message);
    return { total: 0, passed: 0, failed: 0, percentage: 0 };
  }
}

async function collectCoverageResults() {
  try {
    const output = execSync('npm run test:coverage', { encoding: 'utf8' });
    
    // 커버리지 결과에서 퍼센티지 추출
    const coverageMatch = output.match(/All files[^|]*\|([^|]*)/);
    if (coverageMatch) {
      const coverage = parseFloat(coverageMatch[1]);
      return { percentage: coverage };
    }
    
    return { percentage: 0 };
  } catch (error) {
    console.error('Failed to collect coverage results:', error.message);
    return { percentage: 0 };
  }
}

function printQualityReport(evaluation) {
  console.log('\nQuality Gate Report');
  console.log('========================');
  console.log(`Overall Score: ${evaluation.overallScore.toFixed(1)}%`);
  console.log(`Status: ${evaluation.passed ? 'PASSED' : 'FAILED'}`);
  console.log('\nGate Details:');
  
  for (const [gateName, gate] of Object.entries(evaluation.gates)) {
    const status = gate.passed ? 'PASS' : 'FAIL';
    console.log(`  ${status} ${gateName}: ${gate.score.toFixed(1)}% (threshold: ${this.gates[gateName].threshold}%)`);
  }
  
  if (evaluation.recommendation) {
    console.log(`\nRecommendation: ${evaluation.recommendation}`);
  }
}

// 스크립트 실행
if (require.main === module) {
  evaluateQualityGate();
}

module.exports = { evaluateQualityGate };
```
